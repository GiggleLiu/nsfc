\usepackage{tikz}
\usepackage{xeCJK}

\setCJKmainfont{STKaiti}

\definecolor{theorycolor}{RGB}{147,112,219}    % Medium purple
\definecolor{methodcolor}{RGB}{100,149,237}    % Cornflower blue

\begin{document}
\begin{tikzpicture}[

    inputbox/.style={box, fill=inputcolor},

]

% Input - User Problem
\node[inputbox] (input) {用户问题\\NP困难问题实例};

\node[systembox, right=3cm of input, minimum width=3.5cm, minimum height=2.5cm] (compiler) {问题规约编译器};

% Output
\node[outputbox, right=1.5cm of dwave] (output) {优化解};

% Arrows - main flow
\draw[arrow] (input) -- node[above, font=\tiny] {问题实例} (compiler);
\draw[arrow] (compiler) -- node[above, font=\tiny] {硬件映射} (rydberg);
\draw[arrow] (compiler) -- (dwave);
\draw[arrow] (rydberg) -- (output);
\draw[arrow] (dwave) -- (output);

% Theory to compiler
\draw[arrow] (theory) -- node[left, font=\tiny] {最优路径} (compiler);

\draw[dasharrow, color=red!70!black] (output.south) -- ++(0,-0.5) -| node[below, pos=0.25, font=\tiny, text=red!70!black] {实验反馈} (llm.south);

% Labels
